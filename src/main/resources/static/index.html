SQL & Inventory Analytics Explorer

Real-time execution of the 5 required SQL queries (JOIN, GROUP BY, HAVING, ORDER BY, COUNT, SUM)

Query 1: Products with Available Stock

Demonstrates JOIN products, categories, inventories WHERE quantity > 0 ORDER BY quantity DESC

SELECT p.id, p.sku, p.name, c.name AS category_name, p.price, i.quantity AS available_stock FROM products p INNER JOIN categories c ON p.category_id = c.id INNER JOIN inventories i ON p.id = i.product_id WHERE i.quantity > 0 ORDER BY i.quantity DESC;

Query 3: Stock by Category (Aggregation)

Demonstrates GROUP BY category_id, name with COUNT(p.id), SUM(i.quantity), AVG(i.quantity), SUM(price * quantity)

SELECT c.id AS category_id, c.name AS category_name, COUNT(p.id) AS total_products, SUM(i.quantity) AS total_stock_quantity, AVG(i.quantity) AS avg_stock_per_product, SUM(p.price * i.quantity) AS category_inventory_valuation FROM categories c LEFT JOIN products p ON c.id = p.category_id LEFT JOIN inventories i ON p.id = i.product_id GROUP BY c.id, c.name ORDER BY total_stock_quantity DESC;

Query 5: Categories Containing More Than N Products

Demonstrates GROUP BY c.id, c.name HAVING COUNT(p.id) >= :minCount

SELECT c.id AS category_id, c.name AS category_name, COUNT(p.id) AS product_count, SUM(i.quantity) AS total_units_in_category, SUM(p.price * i.quantity) AS category_valuation FROM categories c INNER JOIN products p ON c.id = p.category_id INNER JOIN inventories i ON p.id = i.product_id GROUP BY c.id, c.name HAVING COUNT(p.id) >= :minCount ORDER BY product_count DESC;
